def framed_case(source, demonstration, target):
    grid = [[9 if (r+c) % 2 else 5 for c in range(17)] for r in range(19)]
    def frame(box, color):
        for r, c in strict.perimeter_cells(box):
            grid[r][c] = color
    for box, color in (((0, 0, 8, 16), 1), ((10, 0, 18, 16), 4)):
        frame(box, color)
    for top, left, color, interior in (
            (2, 2, 3, source), (2, 10, 3, demonstration),
            (12, 2, 6, target), (12, 10, 6, [[0]*3 for _ in range(3)])):
        frame((top, left, top+4, left+4), color)
        for r, row in enumerate(interior):
            grid[top+1+r][left+1:left+4] = row
    return grid


def invert(grid, color):
    return [[0 if v == color else color for v in row] for row in grid]

def accepted_layout(grid, demo_color=3, target_color=6):
    grid = copy.deepcopy(grid)
    for box, color in (((0,0,8,16), demo_color),
                       ((10,0,18,16), target_color)):
        for r, c in strict.perimeter_cells(box):
            grid[r][c] = color
    for top, left in ((2,2), (2,10), (12,2), (12,10)):
        for r, c in strict.perimeter_cells((top,left,top+4,left+4)):
            grid[r][c] = 8
    return grid
