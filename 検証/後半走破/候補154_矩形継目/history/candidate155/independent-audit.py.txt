def children(body):
 out=[]
 for axis in [0,1]:
  for k in range(min(p[axis] for p in body),max(p[axis] for p in body)):
   x=tuple(p for p in body if p[axis]<=k); y=tuple(p for p in body if p[axis]>k)
   walls=[p for p in body if p[axis] in [k,k+1]]
   face0={p[1-axis] for p in walls if p[axis]==k};face1={p[1-axis] for p in walls if p[axis]==k+1}
   if face0&face1 and all(p[2]==1 for p in walls) and any(p[2]!=1 for p in x) and any(p[2]!=1 for p in y):out.append((x,y))
 return out
def oracle(body):
 start=(tuple(sorted(body)),);seen={start};todo=[start];terminal=set()
 while todo:
  state=todo.pop();successors=[]
  for i,leaf in enumerate(state):
   successors += [tuple(sorted(state[:i]+state[i+1:]+parts)) for parts in children(leaf)]
  if not successors:terminal.add(state)
  for s in successors:
   if s not in seen:seen.add(s);todo.append(s)
 return terminal
