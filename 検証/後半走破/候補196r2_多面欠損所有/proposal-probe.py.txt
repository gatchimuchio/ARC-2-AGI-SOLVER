import sys,json,time,resource
resource.setrlimit(resource.RLIMIT_CPU,(10,10));resource.setrlimit(resource.RLIMIT_AS,(512*1024**2,512*1024**2))
sys.path.insert(0,'arc2-current110/source')
from 接続.ARC2.直交領域所有核 import unique_region
from 接続.ARC2.既存穴輪郭 import EIGHT_DELTAS
examples=json.load(open('arc2-current99/teachers-only/de809cff.json'))['train']+json.load(open('arc2-current99/solver-input-only/de809cff.json'))['test']
for i,e in enumerate(examples):
 a=e['input'];co=sorted({v for row in a for v in row}-{0});regions=[];start=time.time()
 for color in co:
  mask,info=unique_region([[int(v==color) for v in row] for row in a]);print('region',i,color,info,flush=True);regions.append(mask)
 if any(r is None for r in regions):continue
 h,w=len(a),len(a[0]);clean=[[0]*w for _ in a];overlap=[]
 for k,m in enumerate(regions):
  for r in range(h):
   for c in range(w):
    if m[r][c]:
     if clean[r][c]:overlap.append((r,c))
     clean[r][c]=co[k]
 out=[r[:] for r in clean];holes=[(r,c) for r in range(h) for c in range(w) if a[r][c]==0 and clean[r][c]!=0]
 for r,c in holes:
  other=next(v for v in co if v!=clean[r][c])
  for dr,dc in EIGHT_DELTAS:
   rr,cc=r+dr,c+dc
   if 0<=rr<h and 0<=cc<w:out[rr][cc]=other
 for r,c in holes:out[r][c]=8
 print('RESULT',i,'overlap',overlap,'holes',holes,'fit',out==e.get('output'),'time',time.time()-start,flush=True)
 json.dump({'clean':clean,'output':out,'holes':holes,'overlap':overlap},open(f'proposal196-defect-ownership-20261008/probe-{i}.json','w'))
