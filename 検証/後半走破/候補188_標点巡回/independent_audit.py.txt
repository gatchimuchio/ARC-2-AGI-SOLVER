# Exhaustive independent finite transition oracle uses geometry, never kernel next_steps/paint.
def oracle(tip,direction,markers):
 if not markers:return [(tip,direction,())]
 choices=[]
 for point in sorted(markers):
  dr,dc=point[0]-tip[0],point[1]-tip[1]
  if bool(dr)==bool(dc):continue
  d=((dr>0)-(dr<0),(dc>0)-(dc<0))
  if d==(-direction[0],-direction[1]):continue
  if any(p!=point and (p[0]-tip[0])*d[1]==(p[1]-tip[1])*d[0] and 0<(p[0]-tip[0])*d[0]+(p[1]-tip[1])*d[1]<abs(dr)+abs(dc) for p in markers):continue
  choices.append((point,d))
 if not choices:return [(None,None,())]
 out=[]
 for point,d in choices:
  for end,ed,path in oracle(point,d,markers-{point}):out.append((end,ed,(point,)+path))
 return out
